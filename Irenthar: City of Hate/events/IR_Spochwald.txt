country_event = {#Rooks rebel
	id = spw.100
	title = spw.100.t
	desc = spw.100.d
	picture = GFX_report_event_perkavland_sacked
	
	fire_only_once = yes
	is_triggered_only = yes

	immediate = {
		hidden_effect = {
		#add_ideas = GENERIC_civil_war_isolation
		every_state = {
			limit = {
			is_core_of = RKS
			is_owned_by = ROOT
			}
		RKS = {
		transfer_state = PREV
		}
		}
		RKS = {
		inherit_technology = ROOT
		#add_ideas = GENERIC_civil_war_isolation
		}
		if = {
			limit = {
			check_variable = {
    		var = SPW_rooks_loyalty
    		value = -2
    		compare = less_than_or_equals
			}
			check_variable = {
    		var = SPW_rooks_tension
    		value = 5
    		compare = less_than_or_equals
			}
			}
			RKS = {
			set_politics = {
			ruling_party = republic_peasant
			}
			add_popularity = {
				ideology = republic_peasant
				popularity = 0.25
			}
			}
		}
		if = {
			limit = {
			SPW = { has_country_flag = SPW_jager_rooks_protector }
			}
			RKS = {
			SPW_matthias_jager = {
			set_nationality = RKS
			if = {
				limit = {
					can_be_country_leader = yes
				}
				promote_character = monarchy_absolute_ideology
			}
			else = {
				add_country_leader_role = {
					promote_leader = yes
					country_leader = {
						ideology = monarchy_absolute_ideology
						traits = { heroic_commoner }
					}
				}
				}
			}
			set_politics = {
			ruling_party = monarchy_absolute
			}
			add_popularity = {
				ideology = monarchy_absolute
				popularity = 0.25
			}
			}
		}
		transfer_units_fraction= {
			target = RKS
			size = 0.1
			stockpile_ratio = 0.3
			army_ratio = 0.2
			navy_ratio = 0.0
			air_ratio = 0.0
		}
		}
	}
	
	option = {#nuh uh
		name = spw.100.a
		ai_chance = {
			factor = 1
		}
		RKS = {
		declare_war_on = {
  			  target = ROOT
		      type = annex_everything
			}
		}
	}
	
	option = {#Go Halfings!!
		name = spw.100.b
		
		ai_chance = {
			factor = 0
		}
		RKS = { change_tag_from = ROOT }
		hidden_effect = {
			RKS = {
			declare_war_on = {
  			  target = ROOT
		      type = annex_everything
			}
		}
		}
		}

	option = {#let them go...
		name = spw.100.c
		ai_chance = {
			factor = 1
		}
