### EVENTS FOR MAKING THE AI BUILD THINGS SINCE IT'S DUMB

add_namespace = ai_build

country_event = {#Build Naval Bases on Beachheads
	id = ai_build.1
	
	hidden = yes
	
	trigger = {
		is_ai = yes
		any_state = {
			NOT = {
				is_owned_by = ROOT
				any_neighbor_state = {
					is_controlled_by = ROOT
				}
			}
			is_coastal = yes
			ROOT = {
				divisions_in_state = {
					size > 0
					state = PREV
				}
			}
		}
	}
	
	mean_time_to_happen = {
		days = 2
	}
	
	option = {
		random_state = {
			limit = {
				NOT = {
					is_owned_by = ROOT
					any_neighbor_state = {
						is_controlled_by = ROOT
					}
				}
				is_coastal = yes
				any_country = {
					has_war_with = ROOT
					owns_state = PREV
				}
				ROOT = {
					divisions_in_state = {
						size > 0
						state = PREV
					}
				}
			}
			add_building_construction = {
				type = naval_base
				level = 3
				instant_build = no
				province = {
					all_provinces = yes
					limit_to_coastal = yes
				}
			}
		}
	}
}

country_event = { #Logistical Railway Construction
	id = ai_build.4
	
	hidden = yes
	
	trigger = {
		is_ai = yes
		any_controlled_state = {
			OR = {
				supply_node > 0
				naval_base > 0
			}
			is_in_home_area = yes
			NOT = {
				has_railway_connection = {
					start_state = ROOT.capital
					target_state = THIS
				}
			}
		}
	}
	
	mean_time_to_happen = {
		days = 100
	}

	option = {
		random_controlled_state = {
			limit = {
				OR = {
					supply_node > 0
					naval_base > 0
				}
				is_in_home_area = yes
				NOT = {
					has_railway_connection = {
						start_state = ROOT.capital
						target_state = THIS
					}
				}
			}
			build_railway = {
				level = 1
				controller_priority = {
					base = 1
					modifier = {
						NOT = {
							tag = ROOT
						}
						add = -2
					}
				}
				start_state = ROOT.capital
				target_state = THIS
			}
		}
	}
}
