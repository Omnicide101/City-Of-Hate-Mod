	OR = {
		has_idea = STA_free_trade
		has_idea = STA_racist_bullshit

		has_idea = STA_pious_north

	}

#Checks to see if reasserting Absolutism is a good idea
STA_should_buff_absolutism = {
	OR = {
		AND = { #Can afford to and could be useful
			STA_has_puritanism_buffer = yes
			STA_has_republicanism_buffer = yes
			STA_has_theocracy_buffer = yes
			STA_has_aristocracy_buffer = yes
			monarchy_absolute < 1
			has_stability < 1
		}
		AND = {#Need to before we lose control
			monarchy_absolute < 0.25
			has_stability < 0.25
		}
	}
}
