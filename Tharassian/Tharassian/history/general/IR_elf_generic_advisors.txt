#political advisors
every_possible_country = {
	limit = {
		has_idea = elf_country
		is_dynamic_country = no
		NOT = {
			has_allowed_idea_with_traits = { idea = ambitious_noble characters = yes }
		}
	}
	generate_character = {
		token_base = generic_ambitious_noble # token will be TAG_token_base
		advisor = {
			slot = political_advisor
			traits = { ambitious_noble }
			available = {
				NOT = {
					OR = {
						has_government = republic_peasant
						has_government = republic_merchant
					}
				}
			}
			ai_will_do = {
				factor = 0
				modifier = {
					add = 2
					has_political_power > 200
				}
			}
		}
		portraits = { 
			army = { small = GFX_idea_generic_elf }
		}
	}
}
