country_event = {#Clears Civil War NS
	id = ir.political.5
	title = ir.political.5.t
	desc = ir.political.5.d
	picture = GFX_report_event_communist_leader
	
	trigger = {
		has_idea = GENERIC_civil_war_isolation
		has_war = no
	}
	
	mean_time_to_happen = {
		days = 5
	}
	
	immediate = {
		hidden_effect = {
			if = {
				limit = {
					tag = SKA
				}
				every_state = {
					limit = {
						is_core_of = SKP
					}
					remove_core_of = SKP
				}
				every_state = {
					limit = {
						is_core_of = SKR
					}
					remove_core_of = SKR
				}
			}
		}
	}
	
	option = {
		name = ir.political.5.a
		
		remove_ideas = GENERIC_civil_war_isolation
	}
}

country_event = {#Add Northern Plains separatist tags to the Northern Plains Alliance
	id = ir.political.7

		is_in_faction = no
		has_global_flag = GENERIC_northern_plains_alliance_created
	}
	
	mean_time_to_happen = {
		days = 7
	}
	
	option = {
		random_other_country = {
			limit = {
				is_og_northern_plains_state = yes
				is_faction_leader = yes
			}
			add_to_faction = ROOT
		}
	}
