ORD_strengthen_slavery = {
	if = {
		limit = {
			NOT = {
				OR = {
					has_idea = ORD_slavery_1
					has_idea = ORD_slavery_2
					has_idea = ORD_slavery_3
					has_idea = ORD_slavery_4
					has_idea = ORD_slavery_5
				}
			}
		}
		add_ideas = ORD_slavery_1
	}
	if = {
		limit = {
			has_idea = ORD_slavery_4
		}
		swap_ideas = {
			remove_idea = ORD_slavery_4
			add_idea = ORD_slavery_5
		}
	}
	if = {
		limit = {
			has_idea = ORD_slavery_3
		}
		swap_ideas = {
			remove_idea = ORD_slavery_3
			add_idea = ORD_slavery_4
		}
	}	
	if = {
		limit = {
			has_idea = ORD_slavery_2
		}
		swap_ideas = {
			remove_idea = ORD_slavery_2
			add_idea = ORD_slavery_3
		}
	}
	if = {
		limit = {
			has_idea = ORD_slavery_1
		}
		swap_ideas = {
			remove_idea = ORD_slavery_1
			add_idea = ORD_slavery_2
		}
	}
}
