# Divides the Union between allies
ORD_divide_union = {

			}

		}

			}

		}

	}
	if = {

					has_country_flag = ORD_ally_will_fight_UNG

# Creates a Sanctuary State and transfers population from non-Sanctuary states to this one
ORD_create_sanctuary_state = {
	add_dynamic_modifier = {
		modifier = sanctuary_city
	}
	add_manpower = 600
}


# Country effects for creating a Sanctuary State
ORD_sanctuary_state_country_effects = {
	ORD_improve_royal_power_2 = yes
	ORD_worsen_trad_loyalty_3 = yes

#Set the ratios of power between the various factions
set_faction_power_ratios = {
	set_traditionalist_to_royalist_power_ratio = yes
	set_traditionalist_to_sophist_power_ratio = yes
	set_sophist_to_royalist_power_ratio = yes
	set_sophist_to_traditionalist_power_ratio = yes
	set_royalist_to_traditionalist_power_ratio = yes
	set_royalist_to_sophist_power_ratio = yes
}

set_traditionalist_to_royalist_power_ratio = {

	set_variable = {
		trad_royal_power_ratio = ORD_traditionalists_influence
	}
	
	if = {
		limit = {
			check_variable = {
				ORD_moderates_influence < 1
			}
		}
		set_variable = {
			trad_royal_power_ratio = 100 #Can't divide by zero and we don't wanna flip it negative
		}
	}
	
	else = {
		divide_variable = {
			trad_royal_power_ratio = ORD_moderates_influence
		}
	}
}

#Influences of the Loyalty and Power mechanics
set_ordoria_loyalty_power_mechanics = {
	
	#Traditionalists
	set_variable = { #Traditionalist Loyalty gives PP
		var = ORD_trad_loyalty_effect
		value = ORD_traditionalists_loyalty
	}
	
	multiply_variable = {
		var = ORD_trad_loyalty_effect
		value = 0.01
	}
	
	set_variable = { #Traditionalist Power gives factory output
		var = ORD_trad_power_effect
		value = ORD_traditionalists_influence
	}
	
	multiply_variable = {
		var = ORD_trad_power_effect
		value = 0.01
	}
	
	#Faithful
	set_variable = { #Sophist Loyalty gives War Support
		var = ORD_faith_loyalty_effect
		value = ORD_faithful_loyalty
	}
	
	multiply_variable = {
		var = ORD_faith_loyalty_effect
		value = 0.01
	}
	
	set_variable = { #Sophist Power gives surrender limit
		var = ORD_trad_power_effect
		value = ORD_faithful_influence
	}
	
	multiply_variable = {
		var = ORD_faith_power_effect
		value = 0.01
	}
	
	#Moderates
	set_variable = { #Royalist Loyalty gives War Support
		var = ORD_mod_loyalty_effect
		value = ORD_moderates_loyalty
	}
	
	multiply_variable = {
		var = ORD_mod_loyalty_effect
		value = 0.01
	}
	
	set_variable = { #Royalist Power gives surrender limit
		var = ORD_mod_power_effect
		value = ORD_moderates_influence
	}
	
	multiply_variable = {
		var = ORD_mod_power_effect
		value = 0.01
	}
}
