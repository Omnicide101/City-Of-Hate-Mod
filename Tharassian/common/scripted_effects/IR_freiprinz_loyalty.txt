set_loyalty_approximations = {
	
	#Neutral by default unless overwritten
	set_variable = {
		FPT_loyalty_leaning = 0 #Just want to grill
	}
	
	if = {
		limit = {
			set_temp_variable = { temp_unionloyalty = FPT_union_loyalty }
				
			add_to_temp_variable = { temp_unionloyalty = 5 }
			
			check_variable = {
				FPT_stahlhof_loyalty > temp_unionloyalty
			}
		}
		
		set_variable = {
			FPT_loyalty_leaning = -2 #Strongly Stahlhof
		}
		
		clear_variable = temp_unionloyalty
	}
	
	if = {
		limit = {
			set_temp_variable = { temp_unionloyalty = FPT_union_loyalty }
				
			add_to_temp_variable = { temp_unionloyalty = 2 }
			
			check_variable = {
				FPT_stahlhof_loyalty > temp_unionloyalty
			}
			
			set_temp_variable = { temp_unionloyalty = FPT_union_loyalty }
			
			add_to_temp_variable = { temp_unionloyalty = 4 }
			
			check_variable = {
				var = FPT_stahlhof_loyalty
				value = temp_unionloyalty
				compare = less_than_or_equals
			}
		}
		
		set_variable = {
			FPT_loyalty_leaning = -1 #Leaning Stahlhof
		}
		
		clear_variable = temp_unionloyalty
	}
	
	if = {
		limit = {
			set_temp_variable = { temp_stahlhofloyalty = FPT_stahlhof_loyalty }
			
			add_to_temp_variable = { temp_stahlhofloyalty = 2 }
			
			check_variable = {
				FPT_union_loyalty > temp_stahlhofloyalty
			}
			
			set_temp_variable = { temp_stahlhofloyalty = FPT_stahlhof_loyalty }
			
			add_to_temp_variable = { temp_stahlhofloyalty = 4 }
			
			check_variable = {
				var = FPT_union_loyalty
				value = temp_stahlhofloyalty
				compare = less_than_or_equals
			}
		}
		
		set_variable = {
			FPT_loyalty_leaning = 1 #Leaning loyalist
		}
		clear_variable = temp_stahlhofloyalty
	}
	
	if = {
		limit = {
			set_temp_variable = { temp_stahlhofloyalty = FPT_stahlhof_loyalty }
			
			add_to_temp_variable = { temp_stahlhofloyalty = 4 }
			
			check_variable = {
				FPT_union_loyalty > temp_stahlhofloyalty
			}
		}
		
		set_variable = {
			FPT_loyalty_leaning = 2 #Strongly loyalist
		}
		clear_variable = temp_stahlhofloyalty
	}
}
