SPW_check_weisz_willpower = {
	hidden_effect = {
	remove_all_power_balance_modifiers = {
    	id = SPW_power_balance
		}
	}
	if = {
		limit = {
				check_variable = {
				SPW_weisz_willpower < 25
				}
				check_variable = {
				SPW_weisz_willpower < 50
				}
				check_variable = {
				SPW_weisz_willpower < 75
				}
			}
			add_power_balance_modifier = {
    		id = SPW_power_balance    
    		modifier = SPW_bop_willpower_low
			}
	}
	if = {
		limit = {
				check_variable = {
				SPW_weisz_willpower > 25
				}
				check_variable = {
				SPW_weisz_willpower < 50
				}
				check_variable = {
				SPW_weisz_willpower < 75
				}
			}
			add_power_balance_modifier = {
    		id = SPW_power_balance    
    		modifier = SPW_bop_willpower_okay
			}
	}
	if = {
		limit = {
				check_variable = {
				SPW_weisz_willpower > 25
				}
				check_variable = {
				SPW_weisz_willpower > 50
				}
				check_variable = {
				SPW_weisz_willpower < 75
				}
			}
			add_power_balance_modifier = {
    		id = SPW_power_balance    
    		modifier = SPW_bop_willpower_better
			}
	}
	if = {
		limit = {
				check_variable = {
				SPW_weisz_willpower > 25
				}
				check_variable = {
				SPW_weisz_willpower > 50
				}
				check_variable = {
				SPW_weisz_willpower > 75
				}
			}
			add_power_balance_modifier = {
    		id = SPW_power_balance    
    		modifier = SPW_bop_willpower_good
			}
	}
}

SPW_check_irp_defeat = {
	if = {
		limit = {
				check_variable = {
    			var = IRP_equipment_var
    			value = 0
    			compare = less_than_or_equals
				}
				check_variable = {
    			var = IRP_manpower_var
    			value = 0
    			compare = greater_than_or_equals
				}
			}
				SPW = {
				subtract_from_variable = {
					IRP_manpower_var = 250
					}
				}
	}
	if = {
		limit = {
				check_variable = {
    			var = IRP_equipment_var
    			value = 0
    			compare = less_than_or_equals
				}
				check_variable = {
    			var = IRP_manpower_var
    			value = 0
    			compare = less_than_or_equals
				}
			}
				SPW = {
				subtract_from_variable = {
					IRP_health_var = 1
					}
				}
				random_owned_state = {
    				limit = {
    					has_dynamic_modifier = {
    						modifier = GFX_modifier_river_patrol
								}
    				}
    				remove_dynamic_modifier = {
						modifier = GFX_modifier_river_patrol
					}
			}
	}
	if = {
		limit = {
				check_variable = {
    			var = IRP_equipment_var
    			value = 0
    			compare = less_than_or_equals
				}
				check_variable = {
    			var = IRP_manpower_var
    			value = 0
    			compare = less_than_or_equals
				}
				check_variable = {
    			var = IRP_health_var
    			value = 1
    			compare = equals
				}
			}
				SPW = {
				subtract_from_variable = {
					IRP_health_var = 1
					}
				}
				country_event = spw.101
	}
}
