		icon = GFX_decision_hire_perkav_raiders
		
		allowed = {
			always = yes
		}

			num_of_civilian_factories > 5
			if = {
				limit = {

					tooltip = is_capital_tt #broken loc
					is_capital = yes

				is_owned_and_controlled_by = ROOT
			}
		}
		
		visible = {
			OR = {
				has_tech = perkav_raiders
				has_country_flag = MERCS_perkav_raiders_unlocked

			NOT = { has_war_with = PRK }
		}
		
		state_target = yes
		
		target_trigger = {
			FROM = {
				state = ROOT.capital
			}
		}
		
		on_map_mode = map_and_decisions_view
		
		cost = 0
		
		ai_will_do = {
			factor = 1
			modifier = { #Why on Earth would we hire mercs when we're at peace?
				has_war = no
				factor = 0
			}
			modifier = { #Let's not if we are low on factories
				num_of_available_civilian_factories < 23
				factor = 0.5
			}
			modifier = { #Let's not if we are very low on factories
				num_of_available_civilian_factories < 13
				factor = 0.25
			}
			modifier = { #Best to get units if we're at war
				surrender_progress > 0.25
				factor = 2
			}
			modifier = {#Oh god oh fuck
				surrender_progress > 0.5
				factor = 4
			}
		}
		
		complete_effect = {
			custom_effect_tooltip = MERCS_hire_perkav_raiders_effects_tt
			add_to_variable = { num_mercs_hiring = 1 }

				division_template = {
					name = "Perkav Raider Mercenaries"

	MERCS_drakeskins = {

			}
			if = {
				limit = {
					has_completed_focus = ORD_deploy_drakeskins
				}
				custom_effect_tooltip = MERCS_hire_drakeskins_effects_free_tt
			}
			if = {
				limit = {
					NOT = {	
						tag = ORD
					}
					ORD = {
						has_completed_focus = ORD_deploy_drakeskins
					}
				}
				custom_effect_tooltip = MERCS_alsyon_commissioned_tt
			}
			if = {
				limit = {
					any_other_country = {
						has_character = ORD_captain_alsyon
						has_country_flag = MERCS_hiring_drakeskins_flag
					}
				}
				custom_effect_tooltip = MERCS_alsyon_hired_tt
			}
			if = {
				limit = {
					NOT = {
						ORD = {
							has_completed_focus = ORD_deploy_drakeskins
						}
					}
					any_other_country = {
						has_character = ORD_captain_alsyon
						NOT = {
							has_country_flag = MERCS_hiring_drakeskins_flag
						}
					}
				}
				every_possible_country = {
					limit = {
						has_character = ORD_captain_alsyon
					}
					random_character = {
						limit = {
							is_character = ORD_captain_alsyon
						}
						set_nationality = ROOT
					}
				}
			}
			every_state = {
				limit = {
					has_dynamic_modifier = {
						modifier = ORD_drakeskin_guarded_lands
					}
				}
				remove_dynamic_modifier = {
					modifier = ORD_drakeskin_guarded_lands
				}
			}
			hidden_effect = {
				division_template = {
					name = "Drakeskin Mercenaries"
					is_locked = yes
					
					regiments = {
						halberdier = { x = 0 y = 0 }
						halberdier = { x = 0 y = 1 }
						halberdier = { x = 0 y = 2 }
				        halberdier = { x = 1 y = 0 }
						halberdier = { x = 1 y = 1 }
						halberdier = { x = 1 y = 2 }
					}
				}
				capital_scope = {
					create_unit = {
					    division = "name = \"1st Drakeskin Company\" division_template = \"Drakeskin Mercenaries\" start_experience_factor = 1"
					    owner = ROOT
					}
					create_unit = {
					    division = "name = \"2nd Drakeskin Company\" division_template = \"Drakeskin Mercenaries\" start_experience_factor = 1"
					    owner = ROOT
					}
					create_unit = {
					    division = "name = \"3rd Drakeskin Company\" division_template = \"Drakeskin Mercenaries\" start_experience_factor = 1"

					    division = "name = \"5th Drakeskin Company\" division_template = \"Drakeskin Mercenaries\" start_experience_factor = 1"
