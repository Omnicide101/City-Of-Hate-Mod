GENERIC_subject_interactions = {
	
	GENERIC_subject_support_garrison = {
		
		icon = GFX_decision_SWI_swiss_military_tradition_campaign
		
		state_target = yes
		
		target_trigger = {
			FROM = {
				state = THIS
			}
		}
		
		allowed = {
			always = yes
		}
		
		available = {
			has_manpower > 4999
			any_other_country = {
				is_subject_of = ROOT
				garrison_manpower_need > 0
			}
		}
		
		visible = {
			FROM = {
				owner = {
					is_subject_of = ROOT
					garrison_manpower_need > 0
				}
				is_capital = yes
			}
		}
		
		on_map_mode = map_only
		
		ai_will_do = {
			factor = 0
			modifier = {
				add = 5
				has_manpower > 10000
				any_other_country = {
					is_subject_of = ROOT
					garrison_manpower_need > 1000
					has_manpower < 10000
				}
			}
		}
		
		complete_effect = {
			custom_effect_tooltip = GENERIC_support_subject_garrison_tt
			if = {
				limit = {
					has_manpower < 50000
				}
				add_manpower = -5000
				FROM.owner = {
					add_manpower = 5000
				}
			}
			else = {
				set_temp_variable = {
					sent_manpower = var:manpower_k
				}
				multiply_temp_variable = {
					sent_manpower = 100 #To send 10% of available manpower (manpower_k is in thousands, so 1/1000 * 100 = 1/10)
				}
				FROM.owner = {
					add_manpower = var:sent_manpower
				}
				multiply_temp_variable = {
					sent_manpower = -1
				}
				add_manpower = var:sent_manpower
			}
		}
	}
}
