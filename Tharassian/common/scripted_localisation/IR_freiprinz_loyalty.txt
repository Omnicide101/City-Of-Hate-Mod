# KOnkord Loyalty Show Scripts

defined_text = {
	name = wss_win_loyalty
	text = { 
		trigger = {
			is_subject_of = UNG
			NOT = {
				STA = {
					has_country_flag = STA_civil_war_started
				}
			}
			NOT = { TAG = WIS }
				WIS = {
				has_country_flag = wws_konkord_yes
			}
		}
		localization_key = wss_win_loyalty_show
	}
	text = { 
		trigger = {
			is_subject_of = UNG
			NOT = {
				STA = {
					has_country_flag = STA_civil_war_started
				}
			}
			OR = {
			AND = {
				TAG = WIS
				has_country_flag = wws_konkord_yes
				}
			NOT = {
				WIS = {
				has_country_flag = wws_konkord_yes
				}
				}
			}
		}
		localization_key = wss_win_loyalty_no
	}
}
