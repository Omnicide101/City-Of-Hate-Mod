defined_text = { # Check if Faustburg can stand for elections
	name = GetElectionsFaustburgEligible
	text = {
		trigger = {
			ORD = {
				OR = {
					has_country_flag = ORD_new_UNG_elections_FAU_STA_banned
					has_country_flag = ORD_new_UNG_elections_FAU_banned
				}
			}
		}
		localization_key = loc_key_ungerdaler_elections_faustburg_banned
	}
	text = {
		trigger = {
			NOT = {
				AND = {
					ORD = {
						has_country_flag = ORD_new_UNG_elections_FAU_STA_banned
						has_country_flag = ORD_new_UNG_elections_FAU_banned
					}
				}
			}
		}
		localization_key = loc_key_ungerdaler_elections_faustburg_allowed
	}
}
