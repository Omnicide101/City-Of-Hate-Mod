on_actions = {

	#Daily Ordoria effects
	on_daily_ORD = {
		effect = {
			
			#Calculate power ratios between the various factions
			set_faction_power_ratios = yes
			
			#Calculate steel crisis penalties
			set_temp_variable = {
				ORD_steel_crisis_multiplier_calculator = ORD_steel_crisis_division_limit
			}
			subtract_from_temp_variable = {
				ORD_steel_crisis_multiplier_calculator = num_divisions
			}
			clamp_temp_variable = { #No getting bonuses for having surplus steel, you scamp you
				var = ORD_steel_crisis_multiplier_calculator
				max = 0
			}
			divide_temp_variable = {
				ORD_steel_crisis_multiplier_calculator = 100
			}
			set_variable = {
				ORD_steel_crisis_multiplier_war_support = ORD_steel_crisis_multiplier_calculator
			}
			multiply_temp_variable = {
				ORD_steel_crisis_multiplier_calculator = 2.8
			}
			set_variable = {
				ORD_steel_crisis_multiplier_attack = ORD_steel_crisis_multiplier_calculator
			}
			set_variable = {
				ORD_steel_crisis_multiplier_defence = ORD_steel_crisis_multiplier_calculator
			}
			multiply_temp_variable = {
				ORD_steel_crisis_multiplier_calculator = 2.3
			}
			set_variable = {
				ORD_steel_crisis_multiplier_breakthrough = ORD_steel_crisis_multiplier_calculator
			}
			multiply_temp_variable = {
				ORD_steel_crisis_multiplier_calculator = -1.7
			}
			set_variable = {
				ORD_steel_crisis_multiplier_resource_penalty = ORD_steel_crisis_multiplier_calculator
			}
		}
	}
	
	#Change state of Southmountains
	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					OR = {
						ROOT = {
							tag = ORD
						}
						ROOT = {
							tag = EOR
						}
					}
					OR = {
						FROM.FROM = {
							state = 105
						}
						FROM.FROM = {
							state = 106
						}
						FROM.FROM = {
							state = 1291
						}
					}
				}
				subtract_from_variable = {
					ORD_steel_crisis_division_limit = 5
				}
			}
			if = {
				limit = {
					OR = {
						ROOT = {
							tag = ORD
						}
						ROOT = {
							tag = EOR
						}
					}
					OR = {
						FROM.FROM = {
							state = 105
						}
						FROM.FROM = {
							state = 106
						}
						FROM.FROM = {
							state = 1291
						}
					}
				}
				add_to_variable = {
					ORD_steel_crisis_division_limit = 5
				}
			}
		}
	}
	
	#Crush factions on rebel war end
	on_civil_war_end_before_annexation = {
		effect = {
			if = {
				limit = {
					tag = ORD
					FROM = {
						tag = ORB
						has_government = puritan
					}
				}
				ORD_destroy_trads = yes
			}
			if = {
				limit = {
					tag = ORD
					FROM = {
						tag = ORB
						has_government = monarchy_absolute
					}
				}
				ORD_destroy_mods = yes
			}
			if = {
				limit = {
					tag = ORD
					FROM = {
						tag = ORB
						has_government = theocrat
					}
				}
				ORD_destroy_faiths = yes
			}
			if = {
				limit = {
					tag = ORD
					FROM = {
						tag = ORB
					}
				}
				every_state = {
					limit = {
						is_core_of = ORB
					}
					remove_core_of = ORB
				}
			}
		}
	}
}
