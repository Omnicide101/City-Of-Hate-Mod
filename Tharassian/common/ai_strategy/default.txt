#Always have food
do_not_starve = {
	enable = {
		OR = {
			has_fuel < 100
			fuel_ratio < 0.1
		}
	}
	
	abort_when_not_enabled = yes
	
	ai_strategy = {
		type = fuel_buffer
		value = 1000 #No documentation or usecase on this so this is a test to see if it's an absolute value or a percentage of total capacity
	}
	
	ai_strategy = {
		type = building_target
		id = farm
		value = 1000 #No documentation or usecase on this so this is a test to see if it's an absolute value or a percentage of total capacity
	}
}

#Delete units if you can't afford their upkeep
avoid_famine_by_disbanding_army = {
	enable = {
		OR = {
			has_fuel < 100
			fuel_ratio < 0.1
		}
		OR = {
			AND = {
				num_divisions > 12
				is_major = no
			}
			AND = {
				num_divisions > 48
				is_major = yes
			}
		}
	}
	abort_when_not_enabled = yes
	
	ai_strategy = {
        type = force_build_armies
        value = -100
    }
	
	ai_strategy = {
		type = build_army
		id = cavalry
		value = -99999
	}

	ai_strategy = {
		type = build_army
		id = infantry
		value = -99999
	}
	
	ai_strategy = {
        type = force_build_armies
        value = -99999
    }
}
