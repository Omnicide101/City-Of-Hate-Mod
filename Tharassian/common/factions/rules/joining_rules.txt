joining_rule_non_monarchy = {
	type = joining_rules
	visible = {
		NOT = { is_monarchy = yes }
	}
	ai_will_do = {
		base = 0
		modifier = {
			add = 1
			NOT = { is_monarchy = yes }
		}
		modifier = {
			add = 1
			has_government = republic_peasant
		}
	}
	trigger = {
		NOT = {
			is_monarchy = yes
		}
		IF = {
			limit = { NOT = { has_rule = can_join_opposite_factions } }
			has_government = FROM
		}
	}
}
