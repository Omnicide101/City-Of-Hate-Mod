# checks if any country holds a state that is within range of Perkav raiders

perkav_raiding_target_exists = {
	
	any_country = {
		NOT = {
			OR = {
				tag = PRK
				is_subject_of = PRK
				is_in_faction_with = PRK
				PRK = {
					is_subject_of = ROOT
				}
			}
		}
		any_state = {
			has_state_flag = PRK_coastal_state_in_raid_range_flag	
			is_owned_by = PREV
		}
	}
}
	

# checks if THIS is in any way fighting alongside or against ROOT

is_fighting_alongside_ROOT = {
	OR = {
		THIS = {
			is_in_faction_with = ROOT
			has_war_together_with = ROOT
			has_guaranteed = ROOT
			NOT = {
				has_war_with = ROOT
			}
		}
		ROOT = {
			NOT = {
				has_volunteers_amount_from = {
					tag = PREV
					count < 1
				}
			}
		}
	}
}

# checks if country exists and is independent - MUST use is_not_independent, cannot use is_independent = no

is_independent = {
	AND = {
		country_exists = THIS
		THIS = {
			is_subject = no
		}
	}
}

is_not_independent = {
	OR = {
		NOT = {
			country_exists = THIS
		}
		THIS = {
			is_subject = yes
		}
	}
}

#Checks if THIS is in sphere of influence of ROOT

is_sphered = {
	OR = {
		THIS = {
			is_subject_of = ROOT
		}
		THIS = {
			is_in_faction_with = ROOT
		}
	}
}

is_not_sphered = {
	AND = {
		NOT = {
			THIS = {
				is_in_faction_with = ROOT
			}
		}		
		THIS = {
			NOT = {
				is_subject_of = ROOT
			}
		}
	}
}

# Checks if country is a threat
is_threat = {
	country_exists = THIS
	THIS = {
		is_not_sphered = yes
	}
}

is_not_threat = {
	OR = {
		NOT = {
			country_exists = THIS
		}
		THIS = {
			is_sphered = yes
		}
		THIS = {
			has_capitulated = yes
		}
	}
}

#State scope
has_ROOT_at_least_1_div_in_current_state_scope = {
	custom_trigger_tooltip = {
		tooltip = at_least_one_division_in_state
		ROOT = {
			divisions_in_state = {
				state = PREV size > 0
			}
		}
	}
}

has_not_initiated_border_incident_with_ROOT = {
	custom_trigger_tooltip = {
		tooltip = not_initiated_border_incident_with_ROOT
		NOT = {
			any_state = {
				check_variable = { ROOT.defender_state_vs_@PREV = id }
			}
		}
	}
}

is_border_conflict_defender_vs_FROM = {
	has_variable = ROOT.defender_state_vs_@FROM
}
