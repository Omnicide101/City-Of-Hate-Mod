### EVENTS FOR MAKING THE AI WORK PROPERLY
# Specifically, AI strategies normally don't accept scoping to anything other than a tag directly
# So you need to individually specify each tag
# Using the 'add_ai_strategy' effect allows you to scope around this, making it a lot less of a hassle

#Note to self: Can't use 'fire_only_one' in this case, since a *single* country getting it prevents any other country from getting it. It's not once per country, it's once total.

add_namespace = ai_strats

country_event = {#Make Gowie want access to everyone's markets
	id = ai_strats.1

	hidden = yes
	
	is_triggered_only = yes

	option = {
		
		GOW = {
			add_ai_strategy = {
				type = equipment_market_trade_desire
				id = ROOT
				value = 200
			}
		}
	}
}
