	id = fpt.10
	title = fpt.10.t
	desc = fpt.10.d
	picture = GFX_report_event_german_reichstag_gathering
	
	is_triggered_only = yes
	
	#Let the most prepared Freiprinztum win
	immediate = {
		if = {
			limit = {
				NOT = {
					any_other_country = {
						OR = {
							check_variable = {
								FPT_kriegsspiele_preparedness > ROOT.FPT_kriegsspiele_preparedness
							}
							check_variable = {
								FPT_kriegsspiele_preparedness = ROOT.FPT_kriegsspiele_preparedness
							}
						}
					}
				}
			}
			set_country_flag = FPT_kriegsspiele_winner
		}
		#Pick a random winner if there's a tie
		if = {
			limit = {
				has_variable = FPT_kriegsspiele_preparedness
				NOT = {
					any_other_country = {
						OR = {
							check_variable = {
								FPT_kriegsspiele_preparedness > ROOT.FPT_kriegsspiele_preparedness
							}
							has_country_flag = FPT_kriegsspiele_winner
						}
					}
				}
				any_other_country = {
					check_variable = {
						FPT_kriegsspiele_preparedness = ROOT.FPT_kriegsspiele_preparedness
					}
				}
			}
			random_country = {
				limit = {
					has_variable = FPT_kriegsspiele_preparedness
					NOT = {
						any_other_country = {
							check_variable = {
								FPT_kriegsspiele_preparedness > ROOT.FPT_kriegsspiele_preparedness
							}
						}
					}
				}
				set_country_flag = FPT_kriegsspiele_winner
			}
		}
	}
	
	option = {#Let's see
		name = fpt.10.a
		if = {
			limit = {
				has_country_flag = FPT_kriegsspiele_winner
			}
			hidden_effect = {
				country_event = {
					id = fpt.11
					days = 1
				}
			}
		}
		if = {
			limit = {
				NOT = {
					has_country_flag = FPT_kriegsspiele_winner
				}
			}
			hidden_effect = {
				country_event = {
					id = fpt.12
					days = 1
				}
			}
		}
	}
}
