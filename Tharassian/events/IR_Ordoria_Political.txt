### RANDOM ORDORIAN POLITICS EVENTS

add_namespace = ord_political

### CIVIL WAR EVENTS - Triggered by a faction having high influence and low loyalty
country_event = { #Glory League Revolts
	id = ord_political.1
	title = ord_political.1.t
	desc = ord_political.1.d

	trigger = {
		tag = ORD
		OR = {
			has_government = monarchy_absolute
			has_government = theocrat
		}
		NOT = {
			has_country_flag = ORD_politics_solidified
		}
		check_variable = {
			ORD_traditionalists_loyalty < 0
		}
	}
	
	mean_time_to_happen = {
		days = 500
	}
	
	immediate = {
		hidden_effect = {
			set_country_flag = RND_recent_random_event
			random_owned_controlled_state = {
				PREV = {
					set_variable = {
						var = ORD_glory_league_riot_state_id
						value = PREV
					}
				}
			}
		}
	}
	
	option = {#Stamp 'em out
		name = ord_political.1.a
		add_popularity = {
			ideology = puritan
			popularity = -0.35
		}
		add_stability = -0.15
		
		
		
		if = {
			limit = {
				
			}
		}
	}
	
	option = {#Defect
		name = ord_political.1.b
	}
}


### GENERIC EVENTS - Can happen regardless of who's in charge

country_event = { #Glory League - Extralegal Violence
	id = ord_political.4
	title = ord_political.4.t
	desc = ord_political.4.d

	trigger = {
		tag = ORD
		OR = {
			has_government = puritan
			has_government = monarchy_absolute
			has_government = theocrat
		}
		NOT = {
			has_country_flag = ORD_politics_solidified
		}
	}
	
	mean_time_to_happen = {
		days = 500
	}
	
	immediate = {
		hidden_effect = {
			set_country_flag = RND_recent_random_event
			random_owned_controlled_state = {
				PREV = {
					set_variable = {
						var = ORD_glory_league_riot_state_id
						value = PREV
					}
				}
			}
		}
	}
	
	option = {#Good
		name = ord_political.4.a
		add_popularity = {
			ideology = puritan
			popularity = 0.02
		}
		add_stability = -0.01
		ORD_improve_trad_loyalty_1 = yes
		ORD_improve_trad_power_1 = yes
	}
	
	option = {#Have them tried in court
		name = ord_political.4.b
		add_stability = 0.02
		add_political_power = -35
		ORD_worsen_trad_loyalty_1 = yes
		ORD_improve_royal_power_1 = yes
		ORD_improve_faith_loyalty_1 = yes
	}
}
