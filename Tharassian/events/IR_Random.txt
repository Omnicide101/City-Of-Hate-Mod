country_event = {#City Demands Privileges
	id = random.9
	title = random.9.t
	desc = random.9.d
	picture = GFX_report_event_canada_treaty
	
	trigger = {
		NOT = {
			has_country_flag = RND_recent_random_event
		}
		any_owned_state = {
			OR = {
				has_state_category = city
				has_state_category = large_city
				has_state_category = metropolis
				has_state_category = megalopolis
			}
			is_capital = no #Seat of government can't ask for autonomy
		}
	}
	
	mean_time_to_happen = {
		days = 3650
	}
	
	immediate = {
		hidden_effect = {
			set_country_flag = RND_recent_random_event
			random_owned_controlled_state = {
				limit = {
					OR = {
						has_state_category = city
						has_state_category = large_city
						has_state_category = metropolis
						has_state_category = megalopolis
					}
					is_capital = no
					NOT = {
						has_dynamic_modifier = {
							modifier = autonomous_region
						}
						has_dynamic_modifier = {
							modifier = upset_populace
						}
					}
				}
				set_state_flag = RND_city_privileges_state_flag
				PREV = {
					set_variable = {
						var = RND_city_privileges_state_id
						value = PREV
					}
				}
			}
		}
	}
	
	option = {#Grant
		name = random.9.a

		random_owned_controlled_state = {
			limit = {
				has_state_flag = RND_city_privileges_state_flag
			}
			add_dynamic_modifier = {
				modifier = autonomous_region
			}
			clr_state_flag = RND_city_privileges_state_flag
		}
		clear_variable = RND_city_privileges_state_id
	}
	
	option = {#Decline
		name = random.9.b
		random_owned_controlled_state = {
			limit = {
				has_state_flag = RND_city_privileges_state_flag
			}
			add_dynamic_modifier = {
				modifier = upset_populace
				days = 50
			}
			clr_state_flag = RND_city_privileges_state_flag
		}
		clear_variable = RND_city_privileges_state_id
	}
}
