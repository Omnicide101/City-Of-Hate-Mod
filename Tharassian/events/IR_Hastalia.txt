country_event = { #The Tristal Catastrophe
	id = has.1
	title = has.1.t
	desc = has.1.d

	fire_only_once = yes
	
	mean_time_to_happen = {
		days = 7
	}
	
	trigger = {
		tag = HAS
		is_in_faction_with = TRS
		any_other_country = {
			OR = {
				has_idea = MASLEAGUE_member_fervent
				has_idea = MASLEAGUE_member_committed
				has_idea = MASLEAGUE_member_wavering
			}
			has_war_with = TRS
		}
		TRS = {
			surrender_progress > 0.5
		}
	}
	
	option = { #Take control
		name = has.1.a
		ai_chance = {
			factor = 5
			modifier = {
				has_government = monarchy_absolute
				factor = 2
			}
		}
		TRS = {
			country_event = {
				id = trs.1
				days = 1
			}
		}
		
		custom_effect_tooltip = HAS_warning_TRS_civil_war_tt
	}
	
	option = { #Demand they shape up or we ship out
		name = has.1.b
		ai_chance = {
			factor = 5
			modifier = {
				has_government = monarchy_constitutional
				factor = 2
			}
		}
		TRS = {
			country_event = {
				id = trs.2
				days = 1
			}
		}
		
		if = {
			limit = {
				TRS = {
					has_stability < 0.7
				}
			}
			custom_effect_tooltip = HAS_warning_TRS_decadence_civil_war_tt
		}
	}
	
	option = { #Deal w/ it
		name = has.1.c
		ai_chance = {
			factor = 0
		}
		add_war_support = -0.13
	}
}
