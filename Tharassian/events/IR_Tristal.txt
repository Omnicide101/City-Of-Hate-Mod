country_event = { #Hastalian Demands We Shape Up!
	id = trs.2
	title = trs.2.t
	desc = trs.2.d

	is_triggered_only = yes
	
	option = { #Acquiesce
		name = trs.2.a
		ai_chance = {
			factor = 5
			modifier = {
				has_stability < 0.7
				factor = 0.5
			}
		}
		
		HAS = {
			country_event = {
				id = has.4
				days = 1
			}
		}
		
		add_political_power = -125
		
		if = {
			limit = {
				has_stability < 0.7
			}
		
			custom_effect_tooltip = TRS_warning_decadence_civil_war_tt
			
			# Revolter tag becomes faction leader, NOT original state
			# and Hastalia thinks it's at war with the original state?
			# so they can't be added to the faction by Hastalia
			# so we're hacking a workaround ig
			hidden_effect = {
				every_other_country = {
					limit = {
						is_in_faction_with = ROOT
					}
					set_country_flag = TRS_faction_member_temp_flag
				}
				dismantle_faction = yes
			}
			
			start_civil_war = {
				ideology = republic_aristocrat #Breakaway state
				ruling_party = monarchy_constitutional
				keep_political_party_members = yes
				size = 0.4
				capital = 478
			}
			
			hidden_effect = {
				create_faction = acuerdo_del_dragon
				every_other_country = {
					limit = {
						has_country_flag = TRS_faction_member_temp_flag
					}
					TRS = {
						add_to_faction = PREV
					}
					clr_country_flag = TRS_faction_member_temp_flag
				}
				
				if = {
					limit = {
						HAS = {
							is_in_faction_with = TRS
						}
					}
					random_other_country = {
						limit = {
							original_tag = TRS
							has_war_with = TRS
						}
						HAS = {
							add_to_war = {
								targeted_alliance = ROOT
								enemy = PREV
								hostility_reason = asked_to_join
							}
						}
					}
				}
			}
		}
		
		remove_ideas = TRS_widespread_corruption
		
	}
	
	option = { #Absolutely not
		name = trs.2.b
		ai_chance = {
			factor = 3
		}
		
		add_political_power = 35
		
		HAS = {
			country_event = {
				id = has.5
				days = 1
			}
		}
		
		effect_tooltip = {
		
			HAS = {
				leave_faction = yes
			}
		
			every_other_country = {
				limit = {
					OR = {
						has_idea = MASLEAGUE_member_fervent
						has_idea = MASLEAGUE_member_committed
						has_idea = MASLEAGUE_member_wavering
					}
					has_war_with = HAS
				}
				white_peace = {
					tag = HAS
					message = HAS_abandon_TRS_peace_tt
				}
			}
		}
	}
}
