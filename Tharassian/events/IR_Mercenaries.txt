add_namespace = merc

country_event = {#Perkav Contract Over
	id = merc.1
	title = merc.1.t
	desc = merc.1.d
	picture = GFX_report_event_romanian_soldiers
	
	immediate = {
		set_country_flag = MERCS_perkav_raiders_contract_already_ending_flag
	}
	
	trigger = {
		has_country_flag = MERCS_hiring_perkav_raiders_flag
		NOT = {
			has_idea = MERCS_hiring_perkav_raiders_idea
		}
		NOT = {
			has_country_flag = MERCS_perkav_raiders_contract_already_ending_flag
		}
	}
	
	is_triggered_only = yes
	
	option = {#Very Well
		name = merc.1.a
		delete_unit_template_and_units = {
			division_template = "Perkav Raider Mercenaries"
		}
		subtract_from_variable = { num_mercs_hiring = 1 }
		clr_country_flag = MERCS_hiring_perkav_raiders_flag
		clr_country_flag = MERCS_perkav_raiders_contract_already_ending_flag
	}
	
	option = {#Renew Contract
		name = merc.1.b
		
		add_timed_idea = {
			idea = MERCS_hiring_perkav_raiders_idea
			days = 183
		}
		hidden_effect = {
			country_event = { id = merc.1 days = 184 }
		}
		clr_country_flag = MERCS_perkav_raiders_contract_already_ending_flag
	}
}

country_event = {#Drakeskin Contract Over
	id = merc.3
	title = merc.3.t
	desc = merc.3.d
	picture = GFX_report_event_romanian_soldiers
	
	immediate = {
		set_country_flag = MERCS_drakeskins_contract_already_ending_flag
	}
	
	trigger = {
		has_country_flag = MERCS_hiring_drakeskins_flag
		NOT = {
			has_idea = MERCS_hiring_drakeskins_idea
		}
		NOT = {
			has_country_flag = MERCS_drakeskins_contract_already_ending_flag
		}
	}
	
	is_triggered_only = yes
	
	option = {#Very Well
		name = merc.3.a
		delete_unit_template_and_units = {
			division_template = "Drakeskin Mercenaries"
		}
		random_character = {
			limit = {
				is_character = ORD_captain_alsyon
			}
			set_nationality = {
				target_country = ADV
				character = ORD_captain_alsyon
			}
		}
		subtract_from_variable = { num_mercs_hiring = 1 }
		clr_country_flag = MERCS_hiring_drakeskins_flag
		clr_country_flag = MERCS_drakeskins_contract_already_ending_flag
		if = {
			limit = {
				NOT = {
					ORD = {
						has_country_flag = ORD_drakeskins_not_guarding_border
					}
				}
			}
			100 = {
				add_dynamic_modifier = {
					modifier = ORD_drakeskin_guarded_lands
				}
			}
			104 = {
				add_dynamic_modifier = {
					modifier = ORD_drakeskin_guarded_lands
				}
			}
			105 = {
				add_dynamic_modifier = {
					modifier = ORD_drakeskin_guarded_lands
				}
			}
			107 = {
				add_dynamic_modifier = {
					modifier = ORD_drakeskin_guarded_lands
				}
			}
			116 = {
				add_dynamic_modifier = {
					modifier = ORD_drakeskin_guarded_lands
				}
			}
		}
	}
	
	option = {#Renew Contract
		name = merc.3.b
		
		if = {
			limit = {
				NOT = {
					has_completed_focus = ORD_deploy_drakeskins
				}
			}
			add_timed_idea = {
				idea = MERCS_hiring_drakeskins_idea
				days = 183
			}
			hidden_effect = { country_event = { id = merc.3 days = 183 } }
		}
		#if = {
		#	limit = {
		#		NOT = {
		#			ORD = {
		#				has_completed_focus = ORD_deploy_drakeskins
		#			}
		#		}
		#		any_other_country = {
		#			has_character = ORD_captain_alsyon
		#			NOT = {
		#				has_country_flag = MERCS_hiring_drakeskins_flag
		#			}
		#		}
		#	}
		#}
		clr_country_flag = MERCS_drakeskins_contract_already_ending_flag
	}
}
